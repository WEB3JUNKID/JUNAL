     outcome: currentOutcome,
     rr: currentOutcome === 'Profit' ? rr : -1,
     pnl: parseFloat(pnl.toFixed(2)),
     reason: document.getElementById('log-reason').value.trim()
   };

   if (!acc.trades) acc.trades = [];
   acc.trades.push(trade);
   acc.balance = parseFloat((acc.balance + pnl).toFixed(2));

   await saveUserData();
   updateDashboard();
   renderJournal();
   renderCalendar();

   // Check Stage Pass Target
   const targetVal = acc.initialBalance * (1 + acc.targetPercent / 100);
   if (acc.balance >= targetVal) {
     document.getElementById('status-modal').classList.add('open');
   }

   // Reset Form
   document.getElementById('log-reason').value = '';
   document.getElementById('log-rr').value = '';
   window.showToast("Trade logged successfully!", "success");
 };

 /* ── DASHBOARD & STATS ── */
 function updateDashboard() {
   renderAccountSwitcher();
   if (!userData.accounts || userData.accounts.length === 0) return;
   
   const acc = userData.accounts[activeAccIndex];
   if (!acc) return;

   const initial = acc.initialBalance || 100000;
   const current = acc.balance || initial;
   const pnl = current - initial;
   const trades = acc.trades || [];
   const wins = trades.filter(t => t.outcome === 'Profit' || t.pnl >= 0);
   const wr = trades.length > 0 ? ((wins.length / trades.length) * 100).toFixed(1) : '0';
   
   const totalRR = trades.reduce((sum, t) => sum + (parseFloat(t.rr) || 0), 0);
   const rptPct = acc.riskPercent || 1;
   const rptVal = (current * rptPct) / 100;
   
   const longs = trades.filter(t => t.direction === 'Long');
   const shorts = trades.filter(t => t.direction === 'Short');
   const longPnl = longs.reduce((s, t) => s + (t.pnl || 0), 0);
   const longRR = longs.reduce((s, t) => s + (parseFloat(t.rr) || 0), 0);
   const shortPnl = shorts.reduce((s, t) => s + (t.pnl || 0), 0);
   const shortRR = shorts.reduce((s, t) => s + (parseFloat(t.rr) || 0), 0);

   const targetPct = acc.targetPercent || 8;
   const targetVal = initial * (1 + targetPct / 100);

   const maxDdPct = acc.maxDdPercent || 10;
   const dailyDdPct = acc.dailyDdPercent || 5;
   
   let peak = initial;
   let maxDdUsed = 0;
   let running = initial;
   trades.forEach(t => {
     running += (t.pnl || 0);
     if (running > peak) peak = running;
     const dd = ((peak - running) / initial) * 100;
     if (dd > maxDdUsed) maxDdUsed = dd;
   });
   
   const todayStr = new Date().toISOString().split('T')[0];
   const todayTrades = trades.filter(t => t.date && t.date.startsWith(todayStr));
   const todayLosses = todayTrades.filter(t => t.pnl < 0).reduce((s, t) => s + Math.abs(t.pnl), 0);
   const dailyDdUsed = (todayLosses / initial) * 100;

   document.getElementById('dash-bal').innerText = formatMoney(current);
   document.getElementById('dash-target').innerText = `Target: ${formatMoney(targetVal)}`;
   
   const pnlEl = document.getElementById('dash-pnl');
   pnlEl.innerText = (pnl >= 0 ? '+' : '') + formatMoney(pnl);
   pnlEl.className = `sv ${pnl > 0 ? 'pos' : pnl < 0 ? 'neg' : 'neu'}`;
   document.getElementById('dash-rr').innerText = `${totalRR.toFixed(2)} RR`;
   
   document.getElementById('dash-wr').innerText = `${wr}%`;
   document.getElementById('dash-total-trades').innerText = `${trades.length} Trades`;
   
   document.getElementById('dash-rpt').innerText = `${rptPct}%`;
   document.getElementById('dash-rpt-dollar').innerText = formatMoney(rptVal);
   
   document.getElementById('dash-long-pnl').innerText = (longPnl >= 0 ? '+' : '') + formatMoney(longPnl);
   document.getElementById('dash-long-rr').innerText = `${longRR.toFixed(2)} RR`;
   document.getElementById('dash-short-pnl').innerText = (shortPnl >= 0 ? '+' : '') + formatMoney(shortPnl);
   document.getElementById('dash-short-rr').innerText = `${shortRR.toFixed(2)} RR`;

   const ddBar = document.getElementById('dd-bar-fill');
   const ddPctUsed = Math.min(100, (maxDdUsed / maxDdPct) * 100);
   ddBar.style.width = `${ddPctUsed}%`;
   ddBar.className = `dd-bar ${ddPctUsed > 80 ? 'danger' : ddPctUsed > 50 ? 'warn' : ''}`;
   document.getElementById('dd-current').innerText = `${maxDdUsed.toFixed(1)}%`;
   document.getElementById('dd-max').innerText = `${maxDdPct}%`;

   const dddBar = document.getElementById('ddd-bar-fill');
   const dddPctUsed = Math.min(100, (dailyDdUsed / dailyDdPct) * 100);
   dddBar.style.width = `${dddPctUsed}%`;
   dddBar.className = `dd-bar ${dddPctUsed > 80 ? 'danger' : dddPctUsed > 50 ? 'warn' : ''}`;
   document.getElementById('ddd-current').innerText = `${dailyDdUsed.toFixed(1)}%`;
   document.getElementById('ddd-max').innerText = `${dailyDdPct}%`;
 }

 /* ── JOURNAL RENDER ── */
 function renderJournal() {
   const tbody = document.getElementById('journal-body');
   const empty = document.getElementById('journal-empty');
   tbody.innerHTML = '';

   if (!userData.accounts || userData.accounts.length === 0) return;
   const acc = userData.accounts[activeAccIndex];
   if (!acc || !acc.trades || acc.trades.length === 0) {
     empty.style.display = 'block';
     return;
   }
   empty.style.display = 'none';

   const sorted = [...acc.trades].reverse();
   sorted.forEach(t => {
     const tr = document.createElement('tr');
     const d = t.date ? new Date(t.date).toLocaleDateString(undefined, { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' }) : 'N/A';
     const isProfit = t.outcome === 'Profit' || t.pnl >= 0;
     
     tr.innerHTML = `
       <td style="color:var(--text-sec); font-size:11px;">${d}</td>
       <td><span class="pair-tag">${t.pair}</span></td>
       <td><span class="dir-badge ${t.direction ? t.direction.toLowerCase() : 'long'}">${t.direction || 'Long'}</span></td>
       <td><span style="color:${isProfit ? 'var(--green)' : 'var(--red)'}; font-weight:600;">${isProfit ? 'Profit' : 'Loss'}</span></td>
       <td class="rr-v ${isProfit ? 'pos' : 'neg'}">${(t.rr > 0 ? '+' : '') + t.rr} R</td>
       <td class="money-v ${isProfit ? 'pos' : 'neg'}">${(t.pnl >= 0 ? '+' : '') + formatMoney(t.pnl)}</td>
       <td class="reason-cell" title="${t.reason || ''}">${t.reason || '—'}</td>
       <td><button class="btn-del" onclick="deleteTrade('${t.id}')">✕</button></td>
     `;
     tbody.appendChild(tr);
   });
 }

 window.deleteTrade = async (tradeId) => {
   if (!confirm("Are you sure you want to delete this trade?")) return;
   const acc = userData.accounts[activeAccIndex];
   if (!acc) return;
   
   acc.trades = acc.trades.filter(t => t.id !== tradeId);
   const netPnl = acc.trades.reduce((s, t) => s + (t.pnl || 0), 0);
   acc.balance = parseFloat((acc.initialBalance + netPnl).toFixed(2));
   
   await saveUserData();
   updateDashboard();
   renderJournal();
   renderCalendar();
   window.showToast("Trade deleted", "info");
 };

 /* ── CALENDAR RENDER ── */
 function renderCalendar() {
   const calBody = document.getElementById('cal-body');
   const monthLbl = document.getElementById('cal-month-label');
   calBody.innerHTML = '';

   const year = currentDate.getFullYear();
   const month = currentDate.getMonth();
   const monthNames = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];
   monthLbl.innerText = `${monthNames[month]} ${year}`;

   const firstDay = new Date(year, month, 1).getDay();
   const daysInMonth = new Date(year, month + 1, 0).getDate();

   const acc = userData.accounts[activeAccIndex];
   const trades = acc ? (acc.trades || []) : [];

   for (let i = 0; i < firstDay; i++) {
     const emptyCell = document.createElement('div');
     emptyCell.className = 'cal-cell';
     emptyCell.style.opacity = '0.2';
     calBody.appendChild(emptyCell);
   }

   const today = new Date();
   for (let day = 1; day <= daysInMonth; day++) {
     const cell = document.createElement('div');
     const isToday = today.getFullYear() === year && today.getMonth() === month && today.getDate() === day;
     cell.className = `cal-cell ${isToday ? 'today' : ''}`;

     const dateStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
     const dayTrades = trades.filter(t => t.date && t.date.startsWith(dateStr));
     const dayPnl = dayTrades.reduce((s, t) => s + (t.pnl || 0), 0);

     let dotsHtml = '<div class="cal-dots">';
     dayTrades.forEach(t => {
       const isWin = t.outcome === 'Profit' || t.pnl >= 0;
       dotsHtml += `<div class="cal-dot ${isWin ? 'profit' : 'loss'}"></div>`;
     });
     dotsHtml += '</div>';

     let pnlHtml = '';
     if (dayTrades.length > 0) {
       const isPos = dayPnl >= 0;
       pnlHtml = `<div class="cal-pnl" style="color:${isPos ? 'var(--green)' : 'var(--red)'}">${isPos ? '+' : ''}${formatMoney(dayPnl)}</div>`;
     }

     cell.innerHTML = `
       <div class="cal-day">${day}</div>
       ${dotsHtml}
       ${pnlHtml}
     `;
     calBody.appendChild(cell);
   }
 }

 window.changeMonth = (delta) => {
   currentDate.setMonth(currentDate.getMonth() + delta);
   renderCalendar();
 };

       });

     } catch(e) { console.error('MetaApi trades fetch error:', e); }

 window.syncMetaApiTrades = async () => {
   window.showToast("Connecting to MetaApi Cloud...", "info");
   const metaAccounts = await fetchMetaApiAccounts();
   if (!metaAccounts || (Array.isArray(metaAccounts) && metaAccounts.length === 0)) {
     window.showToast("No MetaApi accounts found or server unreachable", "error");
     return;
   }

   const accList = Array.isArray(metaAccounts) ? metaAccounts : (metaAccounts.items || []);
   let syncedTradesCount = 0;

   for (const metaAcc of accList) {
     let localAcc = userData.accounts.find(a => a.metaApiId === metaAcc._id || a.name === metaAcc.name);
     if (!localAcc) {
       localAcc = {
         id: 'acc_' + Date.now() + '_' + Math.random().toString(36).substr(2, 4),
         metaApiId: metaAcc._id,
         name: metaAcc.name || `MT Account (${metaAcc.login || 'Live'})`,
         initialBalance: metaAcc.balance || 100000,
         balance: metaAcc.balance || 100000,
         targetPercent: 8,
         maxDdPercent: 10,
         dailyDdPercent: 5,
         riskPercent: 1,
         trades: []
       };
       userData.accounts.push(localAcc);
     } else {
       localAcc.metaApiId = metaAcc._id;
     }

     const tradesData = await fetchMetaApiHistoricalTrades(metaAcc._id);
     if (tradesData) {
       const tradeList = Array.isArray(tradesData) ? tradesData : (tradesData.items || []);
       tradeList.forEach(t => {
         const tradeId = t._id || t.id || t.positionId;
         if (!tradeId) return;
         const exists = localAcc.trades.some(existing => existing.id === tradeId || existing.metaId === tradeId);
         if (!exists) {
           const isProfit = (t.profit || 0) >= 0;
           const riskAmt = (localAcc.balance * (localAcc.riskPercent || 1)) / 100;
           const calculatedRR = riskAmt > 0 ? parseFloat(((t.profit || 0) / riskAmt).toFixed(2)) : (isProfit ? 1 : -1);

           localAcc.trades.push({
             id: tradeId,
             metaId: tradeId,
             date: t.closeTime || t.openTime || new Date().toISOString(),
             pair: (t.symbol || 'CUSTOM').toUpperCase(),
             direction: (t.type === 'DEAL_TYPE_BUY' || t.type === 'BUY' || t.type === 'POSITION_TYPE_BUY') ? 'Long' : 'Short',
             outcome: isProfit ? 'Profit' : 'Loss',
             rr: calculatedRR,
             pnl: parseFloat((t.profit || 0).toFixed(2)),
             reason: t.comment ? `[MetaApi] ${t.comment}` : 'MetaApi Live Execution'
           });
           syncedTradesCount++;
         }
       });
     }

     const netPnl = localAcc.trades.reduce((s, t) => s + (t.pnl || 0), 0);
     localAcc.balance = parseFloat((localAcc.initialBalance + netPnl).toFixed(2));
   }

   await saveUserData();
   updateDashboard();
   renderJournal();
   renderCalendar();

   if (syncedTradesCount > 0) {
     window.showToast(`Synced ${syncedTradesCount} new trade(s) from MetaApi!`, "success");
   } else {
     window.showToast("MetaApi connected. Accounts up to date!", "success");
   }
 };
